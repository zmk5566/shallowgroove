cmake_minimum_required(VERSION 3.22)
project(ShallowGrooveNative VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(SG_JUCE_DIR "" CACHE PATH "Existing JUCE 8.0.6 source checkout")
if(SG_JUCE_DIR)
 add_subdirectory("${SG_JUCE_DIR}" juce EXCLUDE_FROM_ALL)
else()
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
find_path(FAUST_INCLUDE_DIR faust/dsp/interpreter-dsp.h HINTS "$ENV{FAUST_ROOT}/include" /opt/homebrew/opt/faust/include REQUIRED)
find_library(FAUST_LIBRARY NAMES faust HINTS "$ENV{FAUST_ROOT}/lib" /opt/homebrew/opt/faust/lib REQUIRED)
find_path(FAUST_STDLIB_DIR stdfaust.lib HINTS "$ENV{FAUST_ROOT}/share/faust" /opt/homebrew/opt/faust/share/faust REQUIRED)
add_library(ShallowCore STATIC Core.cpp)
target_include_directories(ShallowCore PUBLIC . PRIVATE "${FAUST_INCLUDE_DIR}")
target_compile_definitions(ShallowCore PRIVATE SG_FAUST_LIBRARY_DIR="${FAUST_STDLIB_DIR}")
target_link_libraries(ShallowCore PUBLIC "${FAUST_LIBRARY}")
set_target_properties(ShallowCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
juce_add_console_app(ShallowGrooveEngine PRODUCT_NAME "ShallowGrooveEngine")
target_sources(ShallowGrooveEngine PRIVATE Standalone.cpp)
target_link_libraries(ShallowGrooveEngine PRIVATE ShallowCore juce::juce_audio_devices juce::juce_events)
set_target_properties(ShallowGrooveEngine PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}>")
juce_add_plugin(ShallowGrooveRack COMPANY_NAME ShallowGroove PLUGIN_MANUFACTURER_CODE ShGr PLUGIN_CODE SgRk
 PRODUCT_NAME "ShallowGroove Rack" FORMATS VST3 IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE
 COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Synth)
target_sources(ShallowGrooveRack PRIVATE Plugin.cpp)
target_link_libraries(ShallowGrooveRack PRIVATE ShallowCore juce::juce_audio_utils)
juce_add_plugin(ShallowGrooveFX COMPANY_NAME ShallowGroove PLUGIN_MANUFACTURER_CODE ShGr PLUGIN_CODE SgFx NEEDS_WEBVIEW2 TRUE
 PRODUCT_NAME "ShallowGroove FX" FORMATS VST3 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx)
# Use the same palette for the native plugin and the desktop editor.
set(_studio_css "${CMAKE_CURRENT_SOURCE_DIR}/../public/studio-base.css")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_studio_css}")
file(READ "${_studio_css}" _studio_style)
set(SG_FX_PALETTE "")
foreach(token bg surface panel line muted ink signal)
 string(REGEX MATCH "--${token}: #([0-9a-fA-F]+);" found "${_studio_style}")
 if(NOT found)
  message(FATAL_ERROR "Missing studio colour: ${token}")
 endif()
 string(APPEND SG_FX_PALETTE "inline const juce::Colour ${token}{0xff${CMAKE_MATCH_1}};\n")
endforeach()
configure_file(FxPalette.h.in FxPalette.h @ONLY)
target_include_directories(ShallowGrooveFX PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_sources(ShallowGrooveFX PRIVATE FxPlugin.cpp)
target_link_libraries(ShallowGrooveFX PRIVATE ShallowCore juce::juce_audio_utils)
target_compile_definitions(ShallowGrooveFX PUBLIC JUCE_WEB_BROWSER=1 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
if(WIN32)
 target_compile_definitions(ShallowGrooveFX PUBLIC JUCE_USE_WIN_WEBVIEW2_WITH_STATIC_LINKING=1)
endif()
foreach(t ShallowGrooveEngine ShallowGrooveRack)
 target_compile_definitions(${t} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endforeach()
add_subdirectory(host-test EXCLUDE_FROM_ALL)
enable_testing()
add_executable(ShallowGrooveCoreCheck EXCLUDE_FROM_ALL CoreTests.cpp)
target_link_libraries(ShallowGrooveCoreCheck PRIVATE ShallowCore)
add_test(NAME native-core COMMAND ShallowGrooveCoreCheck "${CMAKE_CURRENT_SOURCE_DIR}/..")
find_program(SG_NODE node)
if(SG_NODE)
 add_test(NAME native-transport COMMAND "${SG_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/cli-test.mjs" $<TARGET_FILE:ShallowGrooveEngine>)
 set_tests_properties(native-transport PROPERTIES TIMEOUT 30)
endif()
if(SG_NODE AND (WIN32 OR (APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")))
 add_test(NAME native-metronome COMMAND "${SG_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/metronome-test.mjs" $<TARGET_FILE:ShallowGrooveEngine>)
 set_tests_properties(native-metronome PROPERTIES TIMEOUT 30)
endif()

if(SG_NODE AND (WIN32 OR (APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")))
 add_test(NAME native-inspector COMMAND "${SG_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/inspector-test.mjs" $<TARGET_FILE:ShallowGrooveEngine>)
 set_tests_properties(native-inspector PROPERTIES TIMEOUT 30)
endif()

# Install only the standalone runtime, independently of the VST3.
file(GLOB _faust_modules "${FAUST_STDLIB_DIR}/*.lib")
install(FILES ${_faust_modules} DESTINATION faust COMPONENT Engine)
foreach(_dir dx7 old embedded)
 if(IS_DIRECTORY "${FAUST_STDLIB_DIR}/${_dir}")
  install(DIRECTORY "${FAUST_STDLIB_DIR}/${_dir}" DESTINATION faust COMPONENT Engine FILES_MATCHING PATTERN "*.lib")
 endif()
endforeach()
if(WIN32)
 get_filename_component(_faust_lib_dir "${FAUST_LIBRARY}" DIRECTORY)
 find_file(FAUST_DLL faust.dll HINTS "${_faust_lib_dir}" "$ENV{FAUST_ROOT}/bin" REQUIRED)
 get_filename_component(FAUST_DLL_DIR "${FAUST_DLL}" DIRECTORY)
 add_custom_command(TARGET ShallowGrooveEngine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${FAUST_DLL}" "$<TARGET_FILE_DIR:ShallowGrooveEngine>")
 install(FILES "${FAUST_DLL}" DESTINATION . COMPONENT Engine)
 set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
 set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Engine)
 include(InstallRequiredSystemLibraries)
 install(TARGETS ShallowGrooveEngine RUNTIME_DEPENDENCY_SET EngineDLLs RUNTIME DESTINATION . COMPONENT Engine)
 install(RUNTIME_DEPENDENCY_SET EngineDLLs
  DIRECTORIES "${FAUST_DLL_DIR}"
  PRE_EXCLUDE_REGEXES "^[Aa][Pp][Ii]-" "^[Ee][Xx][Tt]-"
  POST_EXCLUDE_REGEXES "/[Ss][Yy][Ss][Tt][Ee][Mm]32/"
  RUNTIME DESTINATION . COMPONENT Engine)
else()
 install(TARGETS ShallowGrooveEngine RUNTIME DESTINATION . COMPONENT Engine)
endif()
